def get_parameter(self, path):

    path = path.rstrip(".")

    try:
        if path == "Device":
            return self.data["Device"]

        if path.startswith("Device."):
            path = path[len("Device."):]

        keys = path.split(".")
        node = self.data["Device"]

        for key in keys:
            if isinstance(node, dict) and key in node:
                node = node[key]
            else:
                raise KeyError(
                    f"Parameter not found: Device.{path}"
                )

        return node

    except Exception as e:
        raise Exception(str(e))


def set_parameter(self, path, value):

    path = path.rstrip(".")

    try:
        if not path.startswith("Device."):
            raise ValueError("Parameter must start with Device.")

        path = path[len("Device."):]

        keys = path.split(".")

        if not keys:
            raise ValueError("Invalid parameter path")

        node = self.data["Device"]

        for key in keys[:-1]:
            if isinstance(node, dict) and key in node:
                node = node[key]
            else:
                raise KeyError(
                    f"Parameter not found: Device.{path}"
                )

        parameter_name = keys[-1]

        if parameter_name not in node:
            raise KeyError(
                f"Parameter not found: Device.{path}"
            )

        node[parameter_name] = value

        return {
            "Status": "Success",
            "Parameter": "Device." + path,
            "Value": value
        }

    except Exception as e:

        return {
            "Status": "Failed",
            "Parameter": "Device." + path,
            "Error": str(e)
        }

Your subscriber currently receives:

Inside your MQTT message callback, use this structure.
def on_message(client, userdata, msg):

    print("\n================================")
    print("MQTT REQUEST RECEIVED")
    print("================================")

    print("Topic:", msg.topic)

    payload = msg.payload.decode()

    print("Payload:", payload)

    try:

        request = json.loads(payload)

        print("\nParsed request:")
        print(json.dumps(request, indent=4))

        operation = request.get("messageType")

        print("\nOperation:", operation)

        # =========================
        # GET
        # =========================

        if operation == "GET":

            parameter = request.get("parameter")

            print("GET parameter:", parameter)

            try:

                value = model.get_parameter(parameter)

                response = {
                    "messageType": "GET_RESPONSE",
                    "parameter": parameter,
                    "value": value,
                    "success": True
                }

            except Exception as e:

                response = {
                    "messageType": "GET_RESPONSE",
                    "parameter": parameter,
                    "success": False,
                    "error": str(e)
                }

            client.publish(
                "router/usp/response",
                json.dumps(response)
            )

            print("\n================================")
            print("MQTT RESPONSE")
            print("================================")

            print(json.dumps(response, indent=4))

        # =========================
        # SET
        # =========================

        elif operation == "SET":

            parameter = request.get("parameter")
            value = request.get("value")

            print("SET parameter:", parameter)
            print("SET value:", value)

            try:

                result = model.set_parameter(
                    parameter,
                    value
                )

                success = result.get("Status") == "Success"

                response = {
                    "messageType": "SET_RESPONSE",
                    "parameter": parameter,
                    "value": value,
                    "success": success,
                    "result": result
                }

            except Exception as e:

                response = {
                    "messageType": "SET_RESPONSE",
                    "parameter": parameter,
                    "value": value,
                    "success": False,
                    "error": str(e)
                }

            client.publish(
                "router/usp/response",
                json.dumps(response)
            )

            print("\n================================")
            print("MQTT RESPONSE")
            print("================================")

            print(json.dumps(response, indent=4))

    except Exception as e:

        print("Error processing MQTT message:", e)
