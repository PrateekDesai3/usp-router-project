import subprocess


def run_command(command, default):

    try:

        result = subprocess.run(

            command,

            shell=True,

            capture_output=True,

            text=True,

            timeout=2

        )

        if result.returncode == 0:

            output = result.stdout.strip()

            if output != "":

                return output

    except Exception:

        pass

    return default
