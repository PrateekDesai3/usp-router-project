import copy
import threading
import time


class TR181Operations:

    def __init__(self, model, notify_callback=None):

        self.model = model
        self.notify_callback = notify_callback

        self.subscriptions = {}

        self.previous_values = {}

        self.notify_thread = None

        self.stop_event = threading.Event()

    # ==========================================================
    # PATH NORMALIZATION
    # ==========================================================

    def normalize_path(self, path):

        path = path.strip()

        if path.endswith("."):
            path = path[:-1]

        return path

    # ==========================================================
    # GET SINGLE PARAMETER
    # ==========================================================

    def get(self, path):

        path = self.normalize_path(path)

        try:

            value = self.model.get_parameter(path)

            return {

                "messageType": "GET_RESPONSE",

                "parameter": path,

                "value": value,

                "success": True

            }

        except Exception as e:

            return {

                "messageType": "GET_RESPONSE",

                "parameter": path,

                "success": False,

                "error": str(e)

            }

    # ==========================================================
    # GET BULK
    # ==========================================================

    def get_bulk(self, paths):

        results = []

        for path in paths:

            path = self.normalize_path(path)

            try:

                value = self.model.get_parameter(path)

                results.append({

                    "parameter": path,

                    "value": value,

                    "success": True

                })

            except Exception as e:

                results.append({

                    "parameter": path,

                    "success": False,

                    "error": str(e)

                })

        return {

            "messageType": "GET_BULK_RESPONSE",

            "results": results,

            "success": True

        }

    # ==========================================================
    # SET
    # ==========================================================

    def set(self, path, value):

        path = self.normalize_path(path)

        try:

            old_value = self.model.get_parameter(path)

            result = self.model.set_parameter(

                path,

                value

            )

            return {

                "messageType": "SET_RESPONSE",

                "parameter": path,

                "oldValue": old_value,

                "value": value,

                "success": bool(result)

            }

        except Exception as e:

            return {

                "messageType": "SET_RESPONSE",

                "parameter": path,

                "success": False,

                "error": str(e)

            }

    # ==========================================================
    # SUBSCRIBE
    # ==========================================================

    def subscribe(self, paths, interval=5):

        subscription_id = str(

            len(self.subscriptions) + 1

        )

        self.subscriptions[subscription_id] = {

            "paths": paths,

            "interval": interval,

            "lastCheck": 0,

            "enabled": True

        }

        # Store initial values

        for path in paths:

            try:

                self.previous_values[path] = copy.deepcopy(

                    self.model.get_parameter(path)

                )

            except Exception:

                pass

        # Start monitor thread

        if self.notify_thread is None:

            self.stop_event.clear()

            self.notify_thread = threading.Thread(

                target=self._notification_loop,

                daemon=True

            )

            self.notify_thread.start()

        return {

            "messageType": "SUBSCRIBE_RESPONSE",

            "subscriptionId": subscription_id,

            "paths": paths,

            "interval": interval,

            "success": True

        }

    # ==========================================================
    # UNSUBSCRIBE
    # ==========================================================

    def unsubscribe(self, subscription_id):

        if subscription_id not in self.subscriptions:

            return {

                "messageType": "UNSUBSCRIBE_RESPONSE",

                "subscriptionId": subscription_id,

                "success": False,

                "error": "Subscription not found"

            }

        del self.subscriptions[subscription_id]

        if not self.subscriptions:

            self.stop_event.set()

            self.notify_thread = None

        return {

            "messageType": "UNSUBSCRIBE_RESPONSE",

            "subscriptionId": subscription_id,

            "success": True

        }

    # ==========================================================
    # NOTIFICATION LOOP
    # ==========================================================

    def _notification_loop(self):

        while not self.stop_event.is_set():

            current_time = time.time()

            for subscription_id, subscription in list(

                self.subscriptions.items()

            ):

                if not subscription["enabled"]:

                    continue

                interval = subscription["interval"]

                if (

                    current_time -

                    subscription["lastCheck"]

                ) < interval:

                    continue

                subscription["lastCheck"] = current_time

                self._check_subscription(

                    subscription_id,

                    subscription

                )

            time.sleep(1)

    # ==========================================================
    # CHECK CHANGES
    # ==========================================================

    def _check_subscription(

        self,

        subscription_id,

        subscription

    ):

        for path in subscription["paths"]:

            try:

                current_value = copy.deepcopy(

                    self.model.get_parameter(path)

                )

                previous_value = self.previous_values.get(

                    path

                )

                if current_value != previous_value:

                    notification = {

                        "messageType": "NOTIFY",

                        "event": "ValueChange",

                        "subscriptionId": subscription_id,

                        "parameter": path,

                        "oldValue": previous_value,

                        "newValue": current_value

                    }

                    self.previous_values[path] = copy.deepcopy(

                        current_value

                    )

                    self.publish_notification(

                        notification

                    )

            except Exception as e:

                print(

                    "Notification error:",

                    path,

                    e

                )

    # ==========================================================
    # NOTIFY
    # ==========================================================

    def publish_notification(self, notification):

        if self.notify_callback:

            self.notify_callback(

                notification

            )

        else:

            print(

                "NOTIFY:",

                notification

            )
