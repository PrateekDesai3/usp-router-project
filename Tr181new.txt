import json
import subprocess


class Hosts:

    def __init__(self):
        pass

    def get(self):

        return {

            "Host": self.get_hosts()

        }

    def get_hosts(self):

        hosts = []

        try:

            output = subprocess.check_output(

                ["ip", "-j", "neigh"],

                text=True

            )

            neighbors = json.loads(output)

            for neighbor in neighbors:

                hosts.append({

                    "IPAddress": neighbor.get("dst"),

                    "MACAddress": neighbor.get("lladdr", ""),

                    "Interface": neighbor.get("dev"),

                    "State": neighbor.get("state")

                })

        except Exception as e:

            hosts.append({

                "Error": str(e)

            })

        return hosts

import subprocess


class Services:

    def __init__(self):
        pass

    def get(self):

        return {

            "Service": self.get_services()

        }

    def get_services(self):

        services = []

        try:

            output = subprocess.check_output(

                [
                    "systemctl",
                    "list-units",
                    "--type=service",
                    "--state=running",
                    "--no-pager",
                    "--no-legend"
                ],

                text=True

            )

            for line in output.splitlines():

                parts = line.split()

                if len(parts) >= 4:

                    services.append({

                        "Name": parts[0],

                        "LoadState": parts[1],

                        "ActiveState": parts[2],

                        "SubState": parts[3]

                    })

        except Exception as e:

            services.append({

                "Error": str(e)

            })

        return services

    def start_service(self, service):

        try:

            subprocess.run(

                [

                    "sudo",

                    "systemctl",

                    "start",

                    service

                ],

                check=True

            )

            return True

        except:

            return False

    def stop_service(self, service):

        try:

            subprocess.run(

                [

                    "sudo",

                    "systemctl",

                    "stop",

                    service

                ],

                check=True

            )

            return True

        except:

            return False

    def restart_service(self, service):

        try:

            subprocess.run(

                [

                    "sudo",

                    "systemctl",

                    "restart",

                    service

                ],

                check=True

            )

            return True

        except:

            return False

import subprocess
from datetime import datetime


class Time:

    def __init__(self):
        pass

    def get(self):

        return {

            "CurrentLocalTime": self.get_local_time(),

            "CurrentUTCTime": self.get_utc_time(),

            "TimeZone": self.get_timezone(),

            "NTP": self.get_ntp_status()

        }

    def get_local_time(self):

        return datetime.now().strftime(

            "%Y-%m-%d %H:%M:%S"

        )

    def get_utc_time(self):

        return datetime.utcnow().strftime(

            "%Y-%m-%d %H:%M:%S"

        )

    def get_timezone(self):

        try:

            output = subprocess.check_output(

                [

                    "timedatectl",

                    "show",

                    "-p",

                    "Timezone"

                ],

                text=True

            )

            return output.strip().split("=")[1]

        except:

            return "Unknown"

    def get_ntp_status(self):

        try:

            output = subprocess.check_output(

                [

                    "timedatectl",

                    "show",

                    "-p",

                    "NTPSynchronized"

                ],

                text=True

            )

            return output.strip().split("=")[1]

        except:

            return "Unknown"

    def set_timezone(self, timezone):

        try:

            subprocess.run(

                [

                    "sudo",

                    "timedatectl",

                    "set-timezone",

                    timezone

                ],

                check=True

            )

            return True

        except:

            return False

import subprocess


class SoftwareModules:

    def __init__(self):
        pass

    def get(self):

        return {

            "ExecutionUnit": self.get_packages()

        }

    def get_packages(self):

        packages = []

        try:

            output = subprocess.check_output(

                [

                    "dpkg-query",

                    "-W",

                    "-f=${Package}|${Version}\n"

                ],

                text=True

            )

            for line in output.splitlines():

                parts = line.split("|")

                if len(parts) == 2:

                    packages.append({

                        "Name": parts[0],

                        "Version": parts[1]

                    })

            packages.append({

                "Error": str(e)

            })

        return packages

            subprocess.run(

                [

                    "sudo",

                    "apt",

                    "install",

                    "-y",

                    package

                ],

                check=True

            )

            return True

        except:

            return False

    def uninstall(self, package):

        try:

            subprocess.run(

                [

                    "sudo",

                    "apt",

                    "-y",

                    package

                ],

        except:

            return False

import psutil
import time


class PeriodicStatistics:

    def __init__(self):

        self.start_time = time.time()

    def get(self):

        return {

            "CPU": self.get_cpu(),

            "Memory": self.get_memory(),

            "Disk": self.get_disk(),

            "Network": self.get_network(),

            "Uptime": self.get_uptime()

        }

    def get_cpu(self):

        return {

            "UsagePercent": psutil.cpu_percent(interval=1),

            "CoreCount": psutil.cpu_count(),

            "FrequencyMHz": psutil.cpu_freq().current if psutil.cpu_freq() else 0

        }

    def get_memory(self):

        memory = psutil.virtual_memory()

        return {

            "Total": memory.total,

            "Used": memory.used,

            "Free": memory.available,

            "UsagePercent": memory.percent

        }

    def get_disk(self):

        disk = psutil.disk_usage("/")

        return {

import json
import subprocess


class Bridging:

    def __init__(self):
        pass

    def get(self):

        return {

            "Bridge": self.get_bridges(),

            "Port": self.get_ports()

        }

    def get_bridges(self):

        bridges = []

        try:

            output = subprocess.check_output(

                [

                    "ip",

                    "-j",

                    "link"

                ],

                text=True

            )

            interfaces = json.loads(output)

            for interface in interfaces:

                if interface.get("linkinfo", {}).get("info_kind") == "bridge":

                    bridges.append({

                        "Name": interface.get("ifname"),

                        "Index": interface.get("ifindex"),

                        "State": interface.get("operstate"),

                        "MTU": interface.get("mtu"),

                        "MACAddress": interface.get("address")

                    })

        except Exception as e:

            bridges.append({

                "Error": str(e)

            })

        return bridges

    def get_ports(self):

        ports = []

        try:

            output = subprocess.check_output(

                [

                    "bridge",

                    "link"

                ],

                text=True

            )

            current = {}

            for line in output.splitlines():

                line = line.strip()

                if ":" in line:

                    if current:

                        ports.append(current)

                    parts = line.split(":")

                    current = {

                        "Interface": parts[1].strip()

                    }

                elif "master" in line:

                    words = line.split()

                    if "master" in words:

                        current["Bridge"] = words[

                            words.index("master") + 1

                        ]

            if current:

                ports.append(current)

        except Exception as e:

            ports.append({

                "Error": str(e)

            })

        return ports

    def create_bridge(self, bridge):

                    "bridge"

                    "ip",

                    "link",

                    "set",

                    bridge,

                    "up"

                check=True

            )

            return False

                    "ip",

                    "link",

                    "delete",

                    bridge,

                    "type",

                    "bridge"

                ],

            )

router/neighbor_discovery.py
import json
import subprocess


class NeighborDiscovery:

    def __init__(self):
        pass

    def get(self):

        return {

            "Neighbor": self.get_neighbors()

        }

    def get_neighbors(self):

        neighbors = []

        try:

            output = subprocess.check_output(

                [

                    "ip",

                    "-j",

                    "-6",

                    "neigh"

                ],

                text=True

            )

            data = json.loads(output)

            for neighbor in data:

                neighbors.append(

                    {

                        "IPAddress": neighbor.get("dst"),

                        "MACAddress": neighbor.get("lladdr", ""),

                        "Interface": neighbor.get("dev"),

                        "State": neighbor.get("state")

                    }

                )

        except Exception as e:

            neighbors.append(

                {

                    "Error": str(e)

                }

            )

        return neighbors

    def flush(self):

        try:

            subprocess.run(

                [

                    "sudo",

                    "ip",

                    "-6",

                    "neigh",

                    "flush",

                    "all"

                ],

                check=True

            )

            return True

        except:

            return False
