import socket
import platform
import subprocess
import os


class DeviceInfo:

    def __init__(self):
        pass

    def get(self):

        return {

            "Manufacturer": self.get_manufacturer(),

            "ManufacturerOUI": self.get_manufacturer_oui(),

            "ModelName": self.get_model(),

            "Description": self.get_description(),

            "ProductClass": self.get_product_class(),

            "SerialNumber": self.get_serial(),

            "HardwareVersion": self.get_hardware(),

            "SoftwareVersion": self.get_software(),

            "HostName": self.get_hostname(),

            "UpTime": self.get_uptime()

        }

    def get_manufacturer(self):

        return "Raspberry Pi Foundation"

    def get_manufacturer_oui(self):

        try:

            mac = open("/sys/class/net/wlan0/address").read().strip()

            return mac.replace(":", "")[:6].upper()

        except:

            return ""

    def get_model(self):

        try:

            return open("/proc/device-tree/model").read().strip("\x00")

        except:

            return platform.machine()

    def get_description(self):

        return "Virtual Router running USP Agent"

    def get_product_class(self):

        return "VirtualRouter"

    def get_serial(self):

        try:

            cpuinfo = open("/proc/cpuinfo").read()

            for line in cpuinfo.splitlines():

                if line.startswith("Serial"):

                    return line.split(":")[1].strip()

        except:

            pass

        return ""

    def get_hardware(self):

        return platform.machine()

    def get_software(self):

        return platform.release()

    def get_hostname(self):

        return socket.gethostname()

    def get_uptime(self):

        with open("/proc/uptime") as f:

            return int(float(f.readline().split()[0]))

router/ip.py
import json
import subprocess


class IP:

    def __init__(self):
        pass

    def get(self):

        return {

            "Interface": self.get_interfaces(),

            "IPAddress": self.get_ip_addresses()

        }

    def get_interfaces(self):

        try:

            output = subprocess.check_output(
                ["ip", "-j", "addr"],
                text=True
            )

            data = json.loads(output)

            interfaces = []

            for interface in data:

                interfaces.append({

                    "Name": interface.get("ifname"),

                    "Index": interface.get("ifindex"),

                    "MACAddress": interface.get("address"),

                    "AdminStatus": interface.get("operstate"),

                    "MTU": interface.get("mtu")

                })

            return interfaces

        except Exception as e:

            return {
                "Error": str(e)
            }

    def get_ip_addresses(self):

        try:

            output = subprocess.check_output(
                ["ip", "-j", "addr"],
                text=True
            )

            data = json.loads(output)

            addresses = []

            for interface in data:

                if "addr_info" not in interface:
                    continue

                for address in interface["addr_info"]:

                    addresses.append({

                        "Interface": interface["ifname"],

                        "Family": address.get("family"),

                        "IPAddress": address.get("local"),

                        "PrefixLength": address.get("prefixlen")

                    })

            return addresses

        except Exception as e:

            return {
                "Error": str(e)
            }

router/routing.py
import json
import subprocess


class Routing:

    def __init__(self):
        pass

    def get(self):

        return {

            "Router": self.get_routes()

        }

    def get_routes(self):

        try:

            output = subprocess.check_output(
                ["ip", "-j", "route"],
                text=True
            )

            routes = json.loads(output)

            routing_table = []

            for route in routes:

                routing_table.append({

                    "Destination": route.get("dst", "default"),

                    "Gateway": route.get("gateway", ""),

                    "Interface": route.get("dev", ""),

                    "Protocol": route.get("protocol", ""),

                    "Scope": route.get("scope", ""),

                    "Metric": route.get("metric", 0)

                })

            return routing_table

        except Exception as e:

            return {

                "Error": str(e)

            }

    def add_route(self, destination, gateway, interface):

        try:

            subprocess.run(

                [
                    "sudo",
                    "ip",
                    "route",
                    "add",
                    destination,
                    "via",
                    gateway,
                    "dev",
                    interface
                ],

                check=True

            )

            return True

        except Exception:

            return False

    def delete_route(self, destination):

        try:

            subprocess.run(

                [
                    "sudo",
                    "ip",
                    "route",
                    "del",
                    destination
                ],

                check=True

            )

            return True

        except Exception:

            return False

router/ethernet.py
import os
import json
import subprocess


class Ethernet:

    def __init__(self):
        pass

    def get(self):

        return {

            "Interface": self.get_interfaces()

        }

    def get_interfaces(self):

        try:

            output = subprocess.check_output(
                ["ip", "-j", "link"],
                text=True
            )

            interfaces = json.loads(output)

            ethernet = []

            for interface in interfaces:

                name = interface.get("ifname")

                if name.startswith("eth"):

                    ethernet.append({

                        "Name": name,

                        "MACAddress": interface.get("address"),

                        "MTU": interface.get("mtu"),

                        "State": interface.get("operstate"),

                        "Speed": self.get_speed(name),

                        "Duplex": self.get_duplex(name),

                        "Carrier": self.get_carrier(name),

                        "Statistics": self.get_statistics(name)

                    })

            return ethernet

        except Exception as e:

            return {

                "Error": str(e)

            }

    def get_speed(self, interface):

        path = f"/sys/class/net/{interface}/speed"

        if os.path.exists(path):

            try:
                return int(open(path).read().strip())
            except:
                return 0

        return 0

    def get_duplex(self, interface):

        path = f"/sys/class/net/{interface}/duplex"

        if os.path.exists(path):

            return open(path).read().strip()

        return ""

    def get_carrier(self, interface):

        path = f"/sys/class/net/{interface}/carrier"

        if os.path.exists(path):

            return open(path).read().strip() == "1"

        return False

    def get_statistics(self, interface):

        base = f"/sys/class/net/{interface}/statistics"

        return {

            "BytesSent": self.read(f"{base}/tx_bytes"),

            "BytesReceived": self.read(f"{base}/rx_bytes"),

            "PacketsSent": self.read(f"{base}/tx_packets"),

            "PacketsReceived": self.read(f"{base}/rx_packets"),

            "ErrorsSent": self.read(f"{base}/tx_errors"),

            "ErrorsReceived": self.read(f"{base}/rx_errors")

        }

    def read(self, path):

        if os.path.exists(path):

            return int(open(path).read().strip())

        return 0

router/wifi.py
import json
import subprocess


class WiFi:

    def __init__(self):
        pass

    def get(self):

        return {

            "Radio": self.get_radio(),

            "SSID": self.get_ssid(),

            "AccessPoint": self.get_access_point()

        }

    def get_radio(self):

        try:

            output = subprocess.check_output(
                ["iw", "dev"],
                text=True
            )

            radios = []

            current = {}

            for line in output.splitlines():

                line = line.strip()

                if line.startswith("Interface"):

                    if current:
                        radios.append(current)

                    current = {
                        "Interface": line.split()[1]
                    }

                elif line.startswith("channel"):

                    parts = line.split()

                    current["Channel"] = parts[1]
                    current["Frequency"] = parts[2]

            if current:
                radios.append(current)

            return radios

        except Exception as e:

            return {
                "Error": str(e)
            }

    def get_ssid(self):

        try:

            output = subprocess.check_output(
                ["iwgetid", "-r"],
                text=True
            ).strip()

            return {

                "Name": output

            }

        except Exception:

            return {}

    def get_access_point(self):

        try:

            output = subprocess.check_output(
                ["iw", "dev", "wlan0", "link"],
                text=True
            )

            ap = {}

            for line in output.splitlines():

                line = line.strip()

                if line.startswith("Connected to"):

                    ap["BSSID"] = line.split()[2]

                elif line.startswith("SSID:"):

                    ap["SSID"] = line.replace("SSID:", "").strip()

                elif line.startswith("freq:"):

                    ap["Frequency"] = line.replace("freq:", "").strip()

                elif line.startswith("signal:"):

                    ap["Signal"] = line.replace("signal:", "").strip()

                elif line.startswith("tx bitrate:"):

                    ap["Bitrate"] = line.replace("tx bitrate:", "").strip()

            return ap

        except Exception:

            return {}

import json
import subprocess


class InterfaceStack:

    def __init__(self):
        pass

    def get(self):

        return {

            "Entries": self.get_stack()

        }

    def get_stack(self):

        try:

            output = subprocess.check_output(
                ["ip", "-j", "link"],
                text=True
            )

            interfaces = json.loads(output)

            stack = []

            for interface in interfaces:

                parent = interface.get("link")

                stack.append({

                    "HigherLayer": interface.get("ifname"),

                    "LowerLayer": parent if parent else "Physical",

                    "Index": interface.get("ifindex"),

                    "State": interface.get("operstate")

                })

            return stack

        except Exception as e:

            return {

                "Error": str(e)

            }

import os


class DNS:

    def __init__(self):
        pass

    def get(self):

        return {

            "Client": {

                "Server": self.get_dns_servers(),

                "SearchDomains": self.get_search_domains()

            }

        }

    def get_dns_servers(self):

        servers = []

        try:

            with open("/etc/resolv.conf") as f:

                for line in f:

                    line = line.strip()

                    if line.startswith("nameserver"):

                        servers.append({

                            "IPAddress": line.split()[1]

                        })

        except Exception as e:

            return {

                "Error": str(e)

            }

        return servers

    def get_search_domains(self):

        domains = []

        try:

            with open("/etc/resolv.conf") as f:

                for line in f:

                    line = line.strip()

                    if line.startswith("search"):

                        for domain in line.split()[1:]:

                            domains.append(domain)

        except:

            pass

        return domains

class Firewall:

        return {

            "Enable": self.is_enabled(),

            "Chains": self.get_chains(),

            "Rules": self.get_rules()

        }

    def is_enabled(self):

        try:

            output = subprocess.check_output(

                [
                    "sudo",
                    "iptables",
                    "-L"
                ],

                text=True

            )

            return len(output) > 0

        except:

            return False

    def get_chains(self):

        chains = []

        try:

            output = subprocess.check_output(

                [
                    "sudo",
                    "iptables",
                    "-L"
                ],

                text=True

            )

            for line in output.splitlines():

                if line.startswith("Chain"):

                    parts = line.split()

                    chains.append({

                        "Name": parts[1],

                        "Policy": parts[3].strip(")")

                    })

        except Exception as e:

            chains.append({

                "Error": str(e)

            })

        return chains

        rules = []

        try:

                [
                    "sudo",
                    "iptables",
                    "-S"
                ],

                text=True

            )

            for line in output.splitlines():

                rules.append({

                    "Rule": line

                })

        except Exception as e:

            rules.append({

                "Error": str(e)

            })

        return rules

    def enable(self):

        return True

    def disable(self):

        return True

The error in your screenshot is:
AttributeError: 'USPAgent' object has no attribute 'on_message'
This means Python cannot find the on_message() method inside your USPAgent class.
It happens here:
self.client.on_message = self.on_message
but there is no method named on_message() in the class (or it’s outside the class because of incorrect indentation).
Check these things:
1. Make sure this method exists inside 
USPAgent
def on_message(
    self,
    client,
    userdata,
    msg
):
    print("Message received")
 
⸻
 
2. Check indentation
It must be inside the class.
Correct:
class USPAgent:

    def __init__(self):
        ...

    def on_connect(self, client, userdata, flags, rc):
        ...

    def on_message(self, client, userdata, msg):
        ...
Wrong:
class USPAgent:

    def __init__(self):
        ...

def on_message(self, client, userdata, msg):
    ...
In the wrong version, on_message is outside the class, so self.on_message doesn’t exist.
 
⸻
 
3. Check the method name
It must be exactly:
on_message
Not:
On_Message
onMessage
onmessage
Python is case-sensitive.
