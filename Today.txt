from tr181_defaults import TR181_SAMPLE_DATA


class TR181Model:

    def __init__(self):

        self.data = TR181_SAMPLE_DATA

    def get_model(self):

        return self.data["Device"]

    def get_parameter(self, path):

        try:

            if path.startswith("Device."):
                path = path[len("Device."):]

            node = self.data["Device"]

            for part in path.split("."):

                node = node[part]

            return node

        except Exception:

            return {
                "Error": "Parameter not found"
            }

    def set_parameter(self, path, value):

        try:

            if path.startswith("Device."):
                path = path[len("Device."):]

            keys = path.split(".")

            node = self.data["Device"]

            for key in keys[:-1]:

                node = node[key]

            node[keys[-1]] = value

            return {

                "Status": "Success",

                "Parameter": path,

                "Value": value

            }

        except Exception as e:

            return {

                "Status": "Failed",

                "Error": str(e)

            }

    def operate(self, command, arguments=None):

        if command == "Reboot":

            return {

                "Status": "Success",

                "Message": "Device Reboot Simulated"

            }

        elif command == "FactoryReset":

            return {

                "Status": "Success",

                "Message": "Factory Reset Simulated"

            }

        elif command == "Ping":

            return {

                "Status": "Success",

                "Target": arguments,

                "Result": "Reachable"

            }

        return {

            "Status": "Unsupported",

            "Command": command

        }

    def notify(self):

        return self.data["Device"]
